% ---------- Custom commands for RTL file paths ----------
% Displays the RTL path text exactly as written, but hyperlinks it to the
% corresponding file on GitHub (bsc-loca/sargantana, main branch). Styled
% blue + underlined like a normal hyperlink. \uline (from ulem) is used
% instead of the LaTeX-kernel \underline because it breaks correctly across
% lines -- important since long RTL paths need to wrap inside table cells.
% \nolinkurl (from hyperref) is used for the displayed text instead of plain
% \url: with colorlinks=true, hyperref makes \url itself a colored link using
% urlcolor, which silently overrode our \textcolor{blue} and left the link
% TEXT black (only the \uline rule stayed blue). \nolinkurl formats identically
% (monospace, breaks at "/" and "_") but creates no link/color of its own, so
% our outer \textcolor applies correctly to both text and underline.
\urlstyle{tt}
% \rtl{path}          -- path text doubles as both the displayed text and the
%                         URL (used when the written path is already correct).
\newcommand{\rtl}[1]{\href{\rtlrepobase#1}{\textcolor{blue}{\uline{\nolinkurl{#1}}}}}
% \rtlx{text}{path}   -- displays "text" exactly as written, but links to the
%                         verified correct "path" (used when the written text
%                         is a shorthand/partial path that doesn't match the
%                         real repo layout).
\newcommand{\rtlx}[2]{\href{\rtlrepobase#2}{\textcolor{blue}{\uline{\nolinkurl{#1}}}}}
% \rtldir{path}       -- for directories: links to the GitHub tree view
%                         instead of a blob view.
\newcommand{\rtldir}[1]{\href{\rtltreebase#1}{\textcolor{blue}{\uline{\nolinkurl{#1}}}}}
% \rtlnolink{path}    -- for a path mentioned in the text that does not
%                         actually exist in the repo (e.g. rob.sv, which the
%                         text explicitly notes is absent) -- styled like code
%                         but not hyperlinked, so it doesn't point to a 404.
\newcommand{\rtlnolink}[1]{\textit{\nolinkurl{#1}}}

  \end{quote}
}
